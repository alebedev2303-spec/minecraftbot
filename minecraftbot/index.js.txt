const { Client, GatewayIntentBits } = require('discord.js');

const { Pool } = require('pg'); 
let pool; // Здесь будет храниться подключение к базе данных

if(process.env.DATABASE_URL){ // Проверяем, есть ли переменная окружения от Railway
    pool = new Pool({connectionString: process.env.DATABASE_URL});
}

const client = new Client({
  intents: [
    GatewayIntentBits.Guilds,
    GatewayIntentBits.MessageContent
  ]
});

client.once('ready', () => {
  console.log('🔥 Бот запущен!');
});

async function getPrice(resourceName){
    const res = await pool.query(
        'SELECT * FROM prices WHERE LOWER(resource)=$1',
        [resourceName.toLowerCase()]
    );
    return res.rows.length ? res.rows[0] : null;
}

client.on('messageCreate', async message => {
  if (!message.content.startsWith('!list')) return; 
  let list = '';
  
  if(pool){
      const rows = await pool.query('SELECT * FROM prices ORDER BY resource ASC');
      for(const row of rows){
          list += `${row.resource}: ${row.price} монет (${row.amount} шт)\n`;
      }
  }

  await message.reply({ content: 'Доступные ресурсы:', embeds: [{ description: list }] });
});

client.on('messageCreate', async message => {
  if (!message.content.startsWith('!calc')) return; 

  const args = message.content.slice(6).split(' ');
  const resourceName = args[0].toLowerCase().trim(); 
  const amount = parseFloat(args[1]); // Количество может быть дробным!

  if (!resourceName || isNaN(amount)) {
    await message.reply(
      'Формат команды:\n`!calc <ресурс> <количество>`\n' +
      'Например: `!calc алмазы 32`, `/calc угольная пыль 0.5`.'
    );
    return;
  }

  const priceData = await getPrice(resourceName); 

  if (!priceData) {
    await message.reply(`❗ Не нашёл такой ресурс: \`${resourceName}\`. Проверь список командой \`!list\``);
    return;
  }

  const pricePerUnit = priceData.price / priceData.amount; // Цена за одну единицу
  const totalCost = (pricePerUnit * amount).toString();

  await message.reply(
    `Стоимость **${amount} единиц ${resourceName}**: **${totalCost} монет**`
  );
});

client.login(process.env.DISCORD_TOKEN);
